1. sync sql server ogins

# Define source and target servers
$SourceServer = "SourceServerName"
$TargetServer = "TargetServerName"

# Load SQL Server module
Import-Module SqlServer

# Get logins from the source server
$Logins = Invoke-Sqlcmd -ServerInstance $SourceServer -Query "SELECT name FROM sys.server_principals WHERE type IN ('S', 'U') AND name NOT LIKE '##%'"

# Sync logins to the target server
foreach ($Login in $Logins) {
    $LoginName = $Login.name
    Invoke-Sqlcmd -ServerInstance $TargetServer -Query "IF NOT EXISTS (SELECT * FROM sys.server_principals WHERE name = '$LoginName') CREATE LOGIN [$LoginName] FROM WINDOWS"
}


2. Sync SQL Server Agent Jobs

# Define source and target servers
$SourceServer = "SourceServerName"
$TargetServer = "TargetServerName"

# Export jobs from the source server
$Jobs = Invoke-Sqlcmd -ServerInstance $SourceServer -Query "SELECT name FROM msdb.dbo.sysjobs"

foreach ($Job in $Jobs) {
    $JobName = $Job.name

    # Script out the job
    $JobScript = Invoke-Sqlcmd -ServerInstance $SourceServer -Query "EXEC msdb.dbo.sp_help_job @job_name = '$JobName'"
    
    # Execute the job script on the target server
    Invoke-Sqlcmd -ServerInstance $TargetServer -Query $JobScript
}
